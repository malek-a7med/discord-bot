# ══════════════════════════════════════════════════════════
# الخطوة 1: تثبيت @distube/spotify بأحدث إصدار (يدعم Spotify نصي)
# ══════════════════════════════════════════════════════════
npm install @distube/spotify@latest --save

# ══════════════════════════════════════════════════════════
# الخطوة 2: حذف الحزم اللي مش محتاجها
# ══════════════════════════════════════════════════════════
npm uninstall play-dl spotify-url-info

# ══════════════════════════════════════════════════════════
# الخطوة 3: حذف ملف music-handler.js القديم (legacy - مش بيتستخدم)
# ══════════════════════════════════════════════════════════
rm -f helpers/music-handler.js

# ══════════════════════════════════════════════════════════
# الخطوة 4: تعديل commands/music.js
# التغييرات:
#   1. SpotifyPlugin يشتغل بـ api mode (بيدعم البحث النصي)
#   2. البحث النصي يروح Spotify أولاً، YouTube fallback، SoundCloud آخر
# ══════════════════════════════════════════════════════════
node -e "
import { readFileSync, writeFileSync } from 'fs';

let code = readFileSync('commands/music.js', 'utf8');

// 1) تغيير إعداد SpotifyPlugin يشتغل بـ api mode مع SPOTIFY_CLIENT_ID/SECRET
const oldPlugin = \`new SpotifyPlugin()\`;
const newPlugin = \`new SpotifyPlugin(
      process.env.SPOTIFY_CLIENT_ID && process.env.SPOTIFY_CLIENT_SECRET
        ? { api: { clientId: process.env.SPOTIFY_CLIENT_ID, clientSecret: process.env.SPOTIFY_CLIENT_SECRET } }
        : {}
    )\`;
code = code.replace(oldPlugin, newPlugin);

// 2) استبدال منطق البحث النصي - Spotify أولاً ثم YouTube ثم SoundCloud
const oldSearch = \`      // ─── بحث نصي — YouTube أولاً (yt-dlp)، fallback لـ SoundCloud ───
      let played = false;

      // 1️⃣ YouTube عبر yt-dlp
      try {
        await distube.play(voiceChannel, query, { ...playOptions, searchSources: ['youtube'] });
        played = true;
      } catch (ytErr) {
        console.warn('⚠️ [Music] YouTube فشل، بيجرب SoundCloud...', ytErr.message);
      }

      // 2️⃣ SoundCloud fallback
      if (!played) {
        try {
          await interaction.editReply({ content: '🔊 بيجرب SoundCloud...' });
          await distube.play(voiceChannel, \\\`scsearch:\\\${query}\\\`, playOptions);
          played = true;
        } catch (scErr) {
          console.warn('⚠️ [Music] SoundCloud فشل:', scErr.message);
        }
      }

      if (!played) {
        return interaction.editReply({
          content: \\\`❌ مش لاقي الأغنية دي على أي منصة!\\\\n💡 جرب ترفق رابط مباشر من:\\\\n🎵 Spotify: \\\\\\\`open.spotify.com\\\\\\\`\\\\n▶️ YouTube: \\\\\\\`youtube.com\\\\\\\`\\\\n🔊 SoundCloud: \\\\\\\`soundcloud.com\\\\\\\`\\\`,
        });
      }\`;

const newSearch = \`      // ─── بحث نصي — Spotify أولاً، YouTube fallback، SoundCloud آخر ───
      let played = false;

      // 1️⃣ Spotify عبر SpotifyPlugin (spsearch:)
      if (process.env.SPOTIFY_CLIENT_ID && process.env.SPOTIFY_CLIENT_SECRET) {
        try {
          await distube.play(voiceChannel, \\\`spsearch:\\\${query}\\\`, playOptions);
          played = true;
        } catch (spErr) {
          console.warn('⚠️ [Music] Spotify فشل، بيجرب YouTube...', spErr.message);
        }
      }

      // 2️⃣ YouTube عبر yt-dlp
      if (!played) {
        try {
          await distube.play(voiceChannel, query, { ...playOptions, searchSources: ['youtube'] });
          played = true;
        } catch (ytErr) {
          console.warn('⚠️ [Music] YouTube فشل، بيجرب SoundCloud...', ytErr.message);
        }
      }

      // 3️⃣ SoundCloud fallback أخير
      if (!played) {
        try {
          await interaction.editReply({ content: '🔊 بيجرب SoundCloud...' });
          await distube.play(voiceChannel, \\\`scsearch:\\\${query}\\\`, playOptions);
          played = true;
        } catch (scErr) {
          console.warn('⚠️ [Music] SoundCloud فشل:', scErr.message);
        }
      }

      if (!played) {
        return interaction.editReply({
          content: \\\`❌ مش لاقي الأغنية دي على أي منصة!\\\\n💡 جرب ترفق رابط مباشر من:\\\\n🎵 Spotify: \\\\\\\`open.spotify.com\\\\\\\`\\\\n▶️ YouTube: \\\\\\\`youtube.com\\\\\\\`\\\\n🔊 SoundCloud: \\\\\\\`soundcloud.com\\\\\\\`\\\`,
        });
      }\`;

code = code.replace(oldSearch, newSearch);

writeFileSync('commands/music.js', code);
console.log('✅ commands/music.js اتحدث بنجاح');
" --input-type=module

bash <<'BASH'
set -e

echo "🚀 بدء الحل النهائي لمشكلة Spotify / yt-dlp / ENOENT..."

# ادخل فولدر المشروع
if [ -d "/workspace/discord-bot" ]; then
  cd /workspace/discord-bot
elif [ -d "/home/runner/workspace" ]; then
  cd /home/runner/workspace
else
  echo "❌ مش لاقي فولدر المشروع. افتح Shell من داخل مشروع البوت."
  exit 1
fi

echo "📁 Project path: $(pwd)"

if [ ! -f "commands/music.js" ]; then
  echo "❌ commands/music.js مش موجود."
  exit 1
fi

cp commands/music.js commands/music.js.bak.youtube-web-search-fix
echo "✅ Backup: commands/music.js.bak.youtube-web-search-fix"

python3 <<'PY'
from pathlib import Path

path = Path("commands/music.js")
text = path.read_text(encoding="utf-8")

web_search_func = r"""
async function resolveYoutubeUrlFromWeb(query) {
  const url = new URL('https://www.youtube.com/results');
  url.searchParams.set('search_query', query);

  const res = await fetch(url, {
    headers: {
      'user-agent': 'Mozilla/5.0 (compatible; ZangiBot/1.0; +https://discord.com)',
      accept: 'text/html,application/xhtml+xml',
      'accept-language': 'en-US,en;q=0.9,ar;q=0.8',
    },
  });
  if (!res.ok) throw new Error(`YouTube search رجّع ${res.status}`);

  const html = await res.text();
  const matches = [...html.matchAll(/"videoId":"([a-zA-Z0-9_-]{11})"/g)]
    .map(match => match[1])
    .filter(Boolean);
  const videoId = matches.find((id, index) => matches.indexOf(id) === index);
  if (!videoId) throw new Error('YouTube search مرجعش فيديو واضح');

  return `https://www.youtube.com/watch?v=${videoId}`;
}

"""

# ضيف function بتاع YouTube web search قبل resolveYoutubeUrl
if "async function resolveYoutubeUrlFromWeb(query)" not in text:
    marker = "async function resolveYoutubeUrl(query) {\n"
    if marker not in text:
        raise SystemExit("❌ مش لاقي resolveYoutubeUrl. لازم تكون مطبق إصلاح Spotify fallback السابق.")
    text = text.replace(marker, web_search_func + marker, 1)

# ضيف تجربة web search قبل yt-dlp داخل resolveYoutubeUrl
needle = """async function resolveYoutubeUrl(query) {
  const safeQuery = String(query || '').replace(/\\s+/g, ' ').trim();
  if (!safeQuery) throw new Error('مفيش كلمات بحث صالحة للتشغيل');

"""

insert = """async function resolveYoutubeUrl(query) {
  const safeQuery = String(query || '').replace(/\\s+/g, ' ').trim();
  if (!safeQuery) throw new Error('مفيش كلمات بحث صالحة للتشغيل');

  try {
    return await resolveYoutubeUrlFromWeb(safeQuery);
  } catch (webError) {
    console.warn('⚠️ [Music] YouTube web search فشل، هنجرب yt-dlp:', webError.message);
  }

"""

if "resolveYoutubeUrlFromWeb(safeQuery)" not in text:
    if needle not in text:
        raise SystemExit("❌ شكل resolveYoutubeUrl مختلف عن المتوقع.")
    text = text.replace(needle, insert, 1)

path.write_text(text, encoding="utf-8")
print("✅ تم تطبيق YouTube web-search fallback")
PY

echo "🧪 فحص Syntax..."
node --check commands/music.js

echo "🧪 فحص Git diff..."
git diff --check

echo "💾 Commit..."
git add commands/music.js
git commit -m "fix: search youtube without yt-dlp binary" || echo "ℹ️ مفيش تغييرات جديدة أو الإصلاح كان متطبق."

echo "🔗 ضبط GitHub remote..."
if git remote get-url origin >/dev/null 2>&1; then
  git remote set-url origin https://github.com/malek-a7med/discord-bot.git
else
  git remote add origin https://github.com/malek-a7med/discord-bot.git
fi

echo "⬆️ محاولة push..."
git push -u origin HEAD:main --force || echo "⚠️ الـ push فشل، غالبًا محتاج GitHub login/token من Replit."

echo "♻️ Restart البوت..."
pkill -f "node index.js" || true
sleep 2

nohup node index.js > bot.log 2>&1 &

echo "✅ البوت اشتغل."
echo "📜 راقب اللوج:"
echo "tail -f bot.log"
echo "🎵 جرّب /شغل تاني."
BASH

python3 <<'PY'
from pathlib import Path

path = Path("commands/music.js")
text = path.read_text(encoding="utf-8")

# 1) imports
if "import { execFile } from 'child_process';" not in text:
    marker = "import { sendMusicCard } from '../helpers/music-card.js';\n"
    if marker not in text:
        raise SystemExit("❌ مش لاقي import بتاع music-card.")
    text = text.replace(
        marker,
        marker + "import { execFile } from 'child_process';\nimport { promisify } from 'util';\n",
        1
    )

# 2) constants
if "const execFileAsync = promisify(execFile);" not in text:
    marker = "const SPOTIFY_INTERNAL_LIMIT = 100;\n"
    if marker not in text:
        raise SystemExit("❌ مش لاقي SPOTIFY_INTERNAL_LIMIT. طبق إصلاح Spotify الأول.")
    text = text.replace(
        marker,
        marker + """const execFileAsync = promisify(execFile);
const YTDLP_CANDIDATES = [
  process.env.YTDLP_PATH,
  '/home/runner/workspace/.pythonlibs/bin/yt-dlp',
  '/usr/local/bin/yt-dlp',
  '/usr/bin/yt-dlp',
  'yt-dlp',
].filter(Boolean);
""",
        1
    )

# 3) add yt-dlp resolver functions before playResolvedSpotifyTracks
resolver_block = r"""
async function runYtDlp(args) {
  let lastError;
  for (const bin of YTDLP_CANDIDATES) {
    try {
      const { stdout } = await execFileAsync(bin, args, {
        timeout: 30000,
        maxBuffer: 1024 * 1024,
      });
      return stdout.trim();
    } catch (e) {
      lastError = e;
      if (e.code === 'ENOENT') continue;
    }
  }
  throw new Error(`yt-dlp search فشل: ${lastError?.message || 'yt-dlp مش موجود'}`);
}

async function resolveYoutubeUrl(query) {
  const safeQuery = String(query || '').replace(/\s+/g, ' ').trim();
  if (!safeQuery) throw new Error('مفيش كلمات بحث صالحة للتشغيل');

  const output = await runYtDlp([
    '--no-playlist',
    '--default-search',
    'ytsearch1',
    '--print',
    'webpage_url',
    `ytsearch1:${safeQuery}`,
  ]);

  const url = output.split(/\r?\n/).find(line => /^https?:\/\//.test(line));
  if (!url) throw new Error(`مش لاقي نتيجة تشغيل لـ: ${safeQuery}`);
  return url;
}

async function playInternalSearch(voiceChannel, query, playOptions, metadata = {}) {
  const youtubeUrl = await resolveYoutubeUrl(query);
  return distube.play(voiceChannel, youtubeUrl, {
    ...playOptions,
    metadata: {
      ...metadata,
      backend: 'yt-dlp',
      searchQuery: query,
    },
  });
}

"""

if "async function runYtDlp(args)" not in text:
    marker = "async function playResolvedSpotifyTracks(voiceChannel, tracks, playOptions) {\n"
    if marker not in text:
        raise SystemExit("❌ مش لاقي playResolvedSpotifyTracks.")
    text = text.replace(marker, resolver_block + marker, 1)

# 4) replace ytsearch inside playResolvedSpotifyTracks
old_play = """async function playResolvedSpotifyTracks(voiceChannel, tracks, playOptions) {
  if (!tracks.length) throw new Error('Spotify مرجعش أي أغاني قابلة للتشغيل');
  const limitedTracks = tracks.slice(0, SPOTIFY_INTERNAL_LIMIT);
  for (const [index, track] of limitedTracks.entries()) {
    await distube.play(voiceChannel, `ytsearch:${track.query}`, {
      ...playOptions,
      metadata: {
        source: 'spotify',
        spotifyUrl: track.url,
        spotifyTitle: track.title,
      },
    });
    if (index === 0) await new Promise(resolve => setTimeout(resolve, 350));
  }
  return limitedTracks.length;
}
"""

new_play = """async function playResolvedSpotifyTracks(voiceChannel, tracks, playOptions) {
  if (!tracks.length) throw new Error('Spotify مرجعش أي أغاني قابلة للتشغيل');
  const limitedTracks = tracks.slice(0, SPOTIFY_INTERNAL_LIMIT);
  for (const [index, track] of limitedTracks.entries()) {
    await playInternalSearch(voiceChannel, track.query, playOptions, {
      source: 'spotify',
      spotifyUrl: track.url,
      spotifyTitle: track.title,
    });
    if (index === 0) await new Promise(resolve => setTimeout(resolve, 350));
  }
  return limitedTracks.length;
}
"""

if old_play in text:
    text = text.replace(old_play, new_play, 1)

# 5) fix musicHandler.resolveSource previous ytsearch returns
old_resolve = """    if (sourceType === 'spotify' || sourceType === 'spotify_playlist') {
      const tracks = await resolveSpotifyUrl(query);
      return tracks.map(track => ({ query: `ytsearch:${track.query}`, title: track.title || track.query, requestedBy: userTag }));
    }
    return [{ query: `ytsearch:${query}`, title: query, requestedBy: userTag }];
"""

new_resolve = """    if (sourceType === 'spotify' || sourceType === 'spotify_playlist') {
      const tracks = await resolveSpotifyUrl(query);
      const resolved = [];
      for (const track of tracks) {
        resolved.push({
          query: await resolveYoutubeUrl(track.query),
          title: track.title || track.query,
          requestedBy: userTag,
        });
      }
      return resolved;
    }
    return [{ query: await resolveYoutubeUrl(query), title: query, requestedBy: userTag }];
"""

if old_resolve in text:
    text = text.replace(old_resolve, new_resolve, 1)

# 6) fix fallback in handlePlay
old_fallback = "        await distube.play(voiceChannel, `ytsearch:${query}`, playOptions);"
new_fallback = "        await playInternalSearch(voiceChannel, query, playOptions, { source: 'spotify_text_fallback' });"

if old_fallback in text:
    text = text.replace(old_fallback, new_fallback, 1)

path.write_text(text, encoding="utf-8")
print("✅ تم تطبيق إصلاح NO_EXTRACTOR_PLUGIN")
PY
